<body>

    <div class="card" id="mainCard">
        <h1 id="question">Ты будешь моей валентинкой? 🌹</h1>
        <div class="buttons">
            <button id="yesBtn" onclick="sayYes()">Да! 🥰</button>
            <button id="noBtn" onmouseover="moveButton()" onclick="moveButton()">Нет 😢</button>
        </div>
    </div>

    <script>
        const noBtn = document.getElementById('noBtn');
        const yesBtn = document.getElementById('yesBtn');
        const question = document.getElementById('question');
        const mainCard = document.getElementById('mainCard');

        // Функция, которая заставляет кнопку "Нет" убегать
        function moveButton() {
            // Вычисляем случайные координаты в пределах экрана, учитывая размеры кнопки
            const x = Math.random() * (window.innerWidth - noBtn.offsetWidth - 40) + 20;
            const y = Math.random() * (window.innerHeight - noBtn.offsetHeight - 40) + 20;
            
            // Переносим кнопку из контейнера прямо на экран, чтобы она летала везде
            if (noBtn.style.position !== 'fixed') {
                noBtn.style.position = 'fixed';
            }
            
            noBtn.style.left = `${x}px`;
            noBtn.style.top = `${y}px`;
        }

        // Функция при нажатии на "Да"
        function sayYes() {
            // Меняем текст и убираем кнопки
            question.innerHTML = "Ура! Ты лучше всех! Моё сердце принадлежит тебе! ❤️🎈";
            yesBtn.style.display = 'none';
            noBtn.style.display = 'none';
            
            // Запускаем бесконечный дождь из сердечек
            setInterval(createHeart, 150);
        }

        // Создание летающих сердечек
        function createHeart() {
            const heart = document.createElement('div');
            heart.classList.add('heart');
            heart.innerHTML = '❤️';
            
            // Случайная позиция по горизонтали и разная скорость
            heart.style.left = Math.random() * 100 + 'vw';
            heart.style.animationDuration = Math.random() * 2 + 2 + 's'; 
            
            document.body.appendChild(heart);
            
            // Удаляем сердечко после завершения анимации
            setTimeout(() => {
                heart.remove();
            }, 4000);
        }
    </script>
